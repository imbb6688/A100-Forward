name: A100 Forward Daily

on:
  workflow_dispatch:
  schedule:
    - cron: '35 15 * * 1-5'
      timezone: 'Asia/Shanghai'

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: a100-forward
  cancel-in-progress: false

jobs:
  run-forward:
    runs-on: ubuntu-latest
    timeout-minutes: 180
    environment:
      name: github-pages

    steps:
      - name: Checkout current repository revision
        uses: actions/checkout@v6

      - name: Setup Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.11'

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt

      - name: Prepare clean runtime
        run: |
          sudo rm -rf /mnt/data/A100_complete_results /mnt/data/A100_v5_results /mnt/data/A100_v6_results /mnt/data/A100_v7_results /mnt/data/forward
          sudo rm -f /mnt/data/A100_2020_2026_raw.parquet /mnt/data/A100_adjustment_factors.parquet /mnt/data/hithink_manifest.json
          sudo mkdir -p /mnt/data/A100_complete_results /mnt/data/A100_v5_results /mnt/data/A100_v6_results /mnt/data/A100_v7_results /mnt/data/forward
          sudo chown -R $USER:$USER /mnt/data
          cp -f scripts/*.py /mnt/data/
          echo "Repository commit: $GITHUB_SHA"
          echo "Runtime parquet reader:"
          grep -n "pyarrow.parquet" /mnt/data/parquet_minread.py
          if grep -q "read_footer" /mnt/data/parquet_minread.py; then
            echo "::error::OLD parquet_minread.py detected"
            exit 1
          fi

      - name: Download and validate HiThink full market data
        env:
          HITHINK_FINANCE_API_KEY: ${{ secrets.HITHINK_FINANCE_API_KEY }}
        run: |
          python /mnt/data/download_hithink.py
          test -s /mnt/data/A100_2020_2026_raw.parquet
          test -s /mnt/data/A100_adjustment_factors.parquet
          ls -lh /mnt/data/A100_2020_2026_raw.parquet
          cat /mnt/data/hithink_manifest.json

      - name: Verify full Parquet before feature build
        run: |
          python - <<'PY'
          import pyarrow.parquet as pq
          p="/mnt/data/A100_2020_2026_raw.parquet"
          pf=pq.ParquetFile(p)
          print("rows=", pf.metadata.num_rows)
          print("row_groups=", pf.metadata.num_row_groups)
          print("columns=", pf.schema.names)
          assert pf.metadata.num_rows > 7_000_000, "Full-market history unexpectedly small"
          required={"ts_code","trade_date","open","high","low","close","pre_close","volume","amount"}
          missing=required-set(pf.schema.names)
          assert not missing, f"missing columns: {sorted(missing)}"
          PY

      - name: Build frozen A100 feature chain
        run: |
          cd /mnt/data
          python a100_canonical_prep.py
          python a100_v5_pit_proxy.py
          python a100_v6_feature_prep_forward.py
          python a100_v6_train_score_forward.py
          python a100_v7_forward_ranker.py
          python a100_forward_live.py

      - name: Copy sanitized outputs to Pages
        run: |
          mkdir -p site
          cp /mnt/data/forward/index.html site/index.html
          cp /mnt/data/forward/latest_signal.json site/latest_signal.json
          cp /mnt/data/forward/latest_candidates.csv site/latest_candidates.csv
          cp /mnt/data/hithink_manifest.json site/data_manifest.json

      - name: Configure Pages
        uses: actions/configure-pages@v5

      - name: Upload Pages artifact
        uses: actions/upload-pages-artifact@v4
        with:
          path: site

      - name: Deploy Pages
        uses: actions/deploy-pages@v4
